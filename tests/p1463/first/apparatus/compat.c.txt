#define main p1463_suite_main
#include "probe.c"
#undef main

int main(int argc, char **argv)
{
    if (argc != 3) return 2;
    const char *paths[] = {
        "tests/p1058/sources/closed_then_two.nl",
        "tests/p1058/sources/same_original_calls.nl",
        "tests/p1062/sources/terminal_explicit_return.nl",
        "tests/p1312/positive.nl",
        "tests/p1455/A-enabled.nl",
        "tests/p1455/c996/D-noncoincident-prefix.nl"};
    size_t accepted = 0;
    for (size_t i=0;i<sizeof(paths)/sizeof(*paths);++i) {
        char path[2048], outpath[2048];
        if (snprintf(path,sizeof(path),"%s/%s",argv[1],paths[i])<0 ||
            snprintf(outpath,sizeof(outpath),"%s/%zu.c",argv[2],i)<0) return 2;
        Program p={0};
        if (!load(path,&p)) return 1;
        NLFunctionUnitDiagnostic d={0};
        NLCheckStatus s=define(&p,&d);
        char *out=NULL; size_t length=98765;
        NLNodeCStatus emission=NL_NODE_C_UNSUPPORTED;
        if (s==NL_CHECK_OK) {
            s=nl_semantic_check_expression(p.context,p.call,&p.artifact,NULL);
            if (s==NL_CHECK_OK) {
                if (!p.artifact || nl_checked_proof_class(p.artifact)!=NL_CHECKED_MACHINE_PROVED) return 1;
                emission=nl_checked_c_node(p.artifact,&out,&length);
            }
        }
        if (emission!=NL_NODE_C_OK && (out || length!=98765)) return 1;
        if (out) {
            if (strlen(out)!=length) return 1;
            FILE *f=fopen(outpath,"wb");
            if (!f || fwrite(out,1,length,f)!=length || fclose(f)) return 1;
            ++accepted;
        }
        printf("%zu source=%s check=%d emit=%d bytes=%zu\n",i,paths[i],(int)s,(int)emission,length);
        free(out);destroy(&p);
    }
    return accepted>=4 ? 0 : 1;
}
