class PlanDone(gdb.FinishBreakpoint):
    def __init__(self,source,span,out):
        super().__init__(gdb.newest_frame(),internal=True);self.src=source;self.span=span;self.out=out
    def stop(self):
        try:
            assert int(self.return_value)==0
            body=self.out.dereference();bs=body.dereference()['source'];u=raw(self.src);b=raw(bs)
            assert u[self.span[0]:self.span[1]]==b
            rec={'kind':'actual_plan_created','unit_source':address(self.src),'unit_sha256':hashlib.sha256(u).hexdigest(),'unit_length':len(u),'body_span':self.span,'body':address(body),'body_source':address(bs),'body_sha256':hashlib.sha256(b).hexdigest(),'body_length':len(b)}
            assert rec['unit_sha256']==expected['sources'][source_name]['sha256'];plans.append(rec);records.append(rec)
        except Exception as e:errors.append('plan:'+repr(e));return True
