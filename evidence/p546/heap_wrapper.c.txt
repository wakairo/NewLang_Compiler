/* Measurement-only Linux/glibc link wrapper; never linked into Production. */
#include <malloc.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/resource.h>
void *__real_malloc(size_t);
void *__real_calloc(size_t, size_t);
void *__real_realloc(void *, size_t);
void __real_free(void *);
int newlang_driver_main(int, char **);
static size_t live, peak, calls;
static void record(void *p)
{
    if (p) {
        live += malloc_usable_size(p);
        if (live > peak) peak = live;
    }
    ++calls;
}
void *__wrap_malloc(size_t n) { void *p = __real_malloc(n); record(p); return p; }
void *__wrap_calloc(size_t n, size_t s) { void *p = __real_calloc(n, s); record(p); return p; }
void *__wrap_realloc(void *p, size_t n)
{
    const size_t before = p ? malloc_usable_size(p) : 0;
    void *q = __real_realloc(p, n);
    if (q || !n) { live -= before; record(q); } else ++calls;
    return q;
}
void __wrap_free(void *p) { if (p) live -= malloc_usable_size(p); __real_free(p); }
int main(int argc, char **argv)
{
    const int status = newlang_driver_main(argc, argv);
    fprintf(stderr, "HEAP peak_usable_bytes=%zu remaining_usable_bytes=%zu calls=%zu\n", peak, live, calls);
    return status;
}
